class Solution {
public:
    int oddCells(int m, int n, vector<vector<int>>& indices) {
        vector<int> row(m,0);
        vector<int> col (n,0);
        for(auto& index : indices)
        {
            row[index[0]]++;
            col[index[1]]++;
        }
        int oddRows = 0;
        int oddCols = 0;
        for(int x:row)
        {
            if(x % 2 == 1)
            oddRows++;

        }
        for(int x :col)
        {
            if(x % 2 == 1)
            oddCols ++;
        }
        return oddRows * (n - oddCols)
        +(m - oddRows)*oddCols;
    }
};
